# --- ImportC K&R default-argument promotion not implemented ---
# exe1.c defines `void paramtest(c,i,f,d) ... float f;` old-style (K&R) but calls
# it through a prototype declaring `double`. C requires the float parameter of an
# unprototyped definition to be passed as a promoted double; ImportC does not yet
# do this (see the test's own `#if 0 // TODO ImportC`), so the K&R definition's
# WASM signature keeps an f32 param while callers push f64. Native tolerates the
# mismatch via lax calling conventions; WASM's strict validation rejects it.
runnable/exe1.c

# --- Require collateral exception chaining ---
# A throw while a finally/scope(exit) runs on the exceptional path must chain
# the new exception onto the in-flight one (Throwable.chainTogether) and let
# the ORIGINAL keep propagating (eh.d bug1513, test4 test54). The wasm EH
# lowering rethrows the collateral exception instead, so the catch sees the
# wrong object. Design sketch: wrap the finally span [lpad..ret-1] in a
# catch_all_ref "collision" region; on catch with flag==0, a runtime helper
# chains inFlight[top-1] <- inFlight[top], then throw_ref the parked original.
runnable/eh.d
runnable/test4.d

# --- Require expression-temporary destruction during unwind ---
# Struct temporaries and by-value arguments constructed inside an expression
# must be destructed when a later subexpression throws. Natively this runs
# via OPdctor/OPddtor DWARF landing pads; the wasm backend has no elem-level
# try_table regions, so these cleanups only run on the normal path (same
# known gap as DMD_Win64 — see testassert.d's version(DMD_Win64) carve-out).
runnable/sdtor.d
runnable/test14903.d
runnable/testassert.d
